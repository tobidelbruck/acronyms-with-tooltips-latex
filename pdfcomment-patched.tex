\usepackage{pdfcomment}

\makeatletter
% Completely redefine \pdftooltip to use /Subtype /Square with a 0-width border.
% This avoids fragile string-patching entirely and guarantees the correct behavior.
\ifpdf
  \renewcommand{\pdftooltip}[3][]%
  {%
    \leavevmode%
    \setkeys{pdftooltip}{#1}%
    \ifpc@lopt@disable%
      #2%
    \else%
      \expandafter\pc@encode@PDF{#3}%
      \hspace*{0pt}%
      \settowidth{\pc@box@width}{#2}%
      \settoheight{\pc@box@height}{#2}%
      \settodepth{\pc@box@depth}{#2}%
      % We force a /Subtype /Square geometric vector annotation box here
      \pdfannot width \the\pc@box@width height \the\pc@box@height depth \the\pc@box@depth%
      {%
        /Subtype /Square%
        /Contents (\pc@pdfenc@contents)%
        /Subj (Tooltip)%
        /BS << /W 0 >>%  Width 0 means it is functionally invisible to renderers
        /C [ ]%         Empty color matrix ensures no lines are drawn
      }%
      #2%
    \fi%
    \stepcounter{zref@unique}%
    \gdef\pc@lopt@mathstyle{}%
  }%
\else
  % Fallback handling for XeLaTeX/dvipdfmx engines if needed
  \renewcommand{\pdftooltip}[3][]%
  {%
    \leavevmode%
    \setkeys{pdftooltip}{#1}%
    \ifpc@lopt@disable%
      #2%
    \else%
      \expandafter\pc@encode@PDF{#3}%
      \hspace*{0pt}%
      \settowidth{\pc@box@width}{#2}%
      \settoheight{\pc@box@height}{#2}%
      \settodepth{\pc@box@depth}{#2}%
      \special{pdf: ann width \the\pc@box@width\space height \the\pc@box@height\space depth \the\pc@box@depth%
        <<%
          /Subtype /Square%
          /Contents (\pc@pdfenc@contents)%
          /Subj (Tooltip)%
          /BS << /W 0 >>%
          /C [ ]%
        >>%
      }%
      #2%
    \fi%
    \stepcounter{zref@unique}%
    \gdef\pc@lopt@mathstyle{}%
  }%
\fi
\makeatother
